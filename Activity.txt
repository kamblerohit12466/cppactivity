#include <iostream>
#include <string>

using namespace std;

class Student {
private:
    static int idCounter;  // Static variable to auto-generate student IDs
public:
    int studentId;
    string name;
    float grade;

    // Default constructor initializes studentId and sets default values
    Student() {
        studentId = ++idCounter;  // Auto-increment studentId
        name = "Unknown";
        grade = 0.0;
    }

    // Function to input student details
    void inputDetails() {
        cout << "Enter name for Student " << studentId << ": ";
        cin.ignore();
        getline(cin, name);
        cout << "Enter grade for " << name << ": ";
        cin >> grade;
    }

    // Function to display student details
    void displayDetails() const {
        cout << "Student ID: " << studentId << ", Name: " << name << ", Grade: " << grade << endl;
    }
};

// Initialize static variable
int Student::idCounter = 1000;  // Starting Student ID

int main() {
    int numStudents;

    cout << "Enter the number of students: ";
    cin >> numStudents;

    Student students[numStudents];  // Array of Student objects

    // Input details for each student
    for (int i = 0; i < numStudents; i++) {
        students[i].inputDetails();
    }

    // Display all students' details
    cout << "\nStudent Details:\n";
    for (int i = 0; i < numStudents; i++) {
        students[i].displayDetails();
    }

    return 0;
}
